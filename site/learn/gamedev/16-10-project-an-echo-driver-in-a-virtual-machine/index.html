Project: An Echo Driver in a Virtual Machine

lesson 16.1010 of 10 in milestone 165 min

This project assembles chapter 16 into a complete, working pair: the XenonEcho driver with ping, echo and statistics commands, and its client. It gives the full driver source with every rule from the chapter applied, a build-load-test-unload checklist for the lab VM with Driver Verifier on, a test plan including hostile inputs, and a note on what KMDF would do for you.

The driver

C++
// XenonEcho: an educational software driver. Build in Debug, test-signed; load only in the lab VM.
#include <ntddk.h>
#include "echo_ioctl.h"

namespace
{
    UNICODE_STRING kDeviceName = RTL_CONSTANT_STRING(L"\\Device\\XenonEcho");
    UNICODE_STRING kLinkName   = RTL_CONSTANT_STRING(L"\\??\\XenonEcho");
    constexpr ULONG kMaxEchoBytes{ 4096 };

    struct EchoState
    {
        EchoStats stats;            // updated with interlocked operations
    };
}

extern "C" DRIVER_INITIALIZE DriverEntry;
DRIVER_UNLOAD EchoUnload;
_Dispatch_type_(IRP_MJ_CREATE) _Dispatch_type_(IRP_MJ_CLOSE) DRIVER_DISPATCH EchoCreateClose;
_Dispatch_type_(IRP_MJ_DEVICE_CONTROL) DRIVER_DISPATCH EchoDeviceControl;

static NTSTATUS Complete(PIRP irp, NTSTATUS status, ULONG_PTR information)
{
    irp->IoStatus.Status = status;
    irp->IoStatus.Information = information;
    IoCompleteRequest(irp, IO_NO_INCREMENT);           // exactly once; do not touch irp afterwards
    return status;
}

NTSTATUS EchoCreateClose(PDEVICE_OBJECT, PIRP irp)
{
    return Complete(irp, STATUS_SUCCESS, 0);
}

NTSTATUS EchoDeviceControl(PDEVICE_OBJECT device, PIRP irp)
{
    auto* state{ static_cast<EchoState*>(device->DeviceExtension) };
    const auto& p{ IoGetCurrentIrpStackLocation(irp)->Parameters.DeviceIoControl };
    void* buffer{ irp->AssociatedIrp.SystemBuffer };   // METHOD_BUFFERED for every code below

    switch (p.IoControlCode)
    {
    case IOCTL_ECHO_PING:
        return Complete(irp, STATUS_SUCCESS, 0);

    case IOCTL_ECHO_ECHO:
    {
        const ULONG in{ p.InputBufferLength };
        if (in == 0 || in > kMaxEchoBytes) return Complete(irp, STATUS_INVALID_PARAMETER, 0);
        if (p.OutputBufferLength < in) return Complete(irp, STATUS_BUFFER_TOO_SMALL, 0);
        InterlockedIncrement64(reinterpret_cast<volatile LONG64*>(&state->stats.requests));
        InterlockedAdd64(reinterpret_cast<volatile LONG64*>(&state->stats.bytesEchoed), in);
        return Complete(irp, STATUS_SUCCESS, in);       // the input is already in the system buffer
    }

    case IOCTL_ECHO_STATS:
    {
        if (p.OutputBufferLength < sizeof(EchoStats)) return Complete(irp, STATUS_BUFFER_TOO_SMALL, 0);
        EchoStats copy{};                               // zeroed: no stray bytes leave the kernel
        copy.requests = static_cast<unsigned long long>(InterlockedCompareExchange64(reinterpret_cast<volatile LONG64*>(&state->stats.requests), 0, 0));
        copy.bytesEchoed = static_cast<unsigned long long>(InterlockedCompareExchange64(reinterpret_cast<volatile LONG64*>(&state->stats.bytesEchoed), 0, 0));
        RtlCopyMemory(buffer, &copy, sizeof(copy));
        return Complete(irp, STATUS_SUCCESS, sizeof(copy));
    }

    default:
        return Complete(irp, STATUS_INVALID_DEVICE_REQUEST, 0);
    }
}

void EchoUnload(PDRIVER_OBJECT driver)
{
    IoDeleteSymbolicLink(&kLinkName);                   // reverse order of creation
    if (driver->DeviceObject) IoDeleteDevice(driver->DeviceObject);
    KdPrint(("XenonEcho: unloaded\n"));
}

extern "C" NTSTATUS DriverEntry(PDRIVER_OBJECT driver, PUNICODE_STRING)
{
    PDEVICE_OBJECT device{ nullptr };
    NTSTATUS status{ IoCreateDevice(driver, sizeof(EchoState), &kDeviceName, ECHO_DEVICE_TYPE,
                                    FILE_DEVICE_SECURE_OPEN, FALSE, &device) };
    if (!NT_SUCCESS(status)) return status;
    RtlZeroMemory(device->DeviceExtension, sizeof(EchoState));

    status = IoCreateSymbolicLink(&kLinkName, &kDeviceName);
    if (!NT_SUCCESS(status))
    {
        IoDeleteDevice(device);                          // undo what succeeded (lesson 16.03)
        return status;
    }

    driver->MajorFunction[IRP_MJ_CREATE] = EchoCreateClose;
    driver->MajorFunction[IRP_MJ_CLOSE] = EchoCreateClose;
    driver->MajorFunction[IRP_MJ_DEVICE_CONTROL] = EchoDeviceControl;
    driver->DriverUnload = EchoUnload;
    KdPrint(("XenonEcho: loaded\n"));
    return STATUS_SUCCESS;
}

For production you would create the device with IoCreateDeviceSecure and an SDDL string limiting access (lesson 16.04); in the lab, testing from an administrator prompt, the plain call keeps the listing short. The client is lesson 16.06's program.

Checklist

  1. Build both projects (x64, Debug). The driver build test-signs XenonEcho.sys (lesson 16.09). Run Code Analysis and fix every warning.
  2. Revert the VM to the clean snapshot (lesson 15.08). Connect WinDbg; set ed nt!Kd_DEFAULT_Mask 0xF.
  3. Copy XenonEcho.sys and EchoClient.exe into the VM.
  4. Enable Driver Verifier for the driver (administrator prompt in the VM), and restart: verifier /standard /driver XenonEcho.sys
  5. Load: sc create XenonEcho type= kernel binPath= C:\Drivers\XenonEcho.sys then sc start XenonEcho. WinDbg shows "XenonEcho: loaded".
  6. Test with the client, then the hostile tests below.
  7. Unload: sc stop XenonEcho, check "unloaded" appears and !poolused shows nothing left for your tag. sc delete XenonEcho.
  8. Revert the VM.

Test plan

Test Expect
Ping Success
Echo "hello, kernel" The same text back, returned = 13
Echo 0 bytes ERROR_INVALID_PARAMETER
Echo 4097 bytes ERROR_INVALID_PARAMETER
Echo 100 bytes into a 50-byte output ERROR_INSUFFICIENT_BUFFER
Stats into a 4-byte output ERROR_INSUFFICIENT_BUFFER
Stats after 3 echoes requests 3, bytes the sum
An undefined code (function 0x900) ERROR_INVALID_FUNCTION
Echo from 8 threads at once, 10,000 times each Correct totals (interlocked statistics)
sc stop while the client holds the device open Fails until the client closes it
Random codes and lengths for a minute (lesson 17.03's fuzzing) No bug check, no Verifier violation

All of this with Driver Verifier on. A pass means the driver behaves correctly on every path you can reach from user mode.

What KMDF would do for you

The same driver in KMDF (lesson 15.05) would replace the dispatch table with I/O queues (WdfIoQueueCreate), get request buffers with checked helpers (WdfRequestRetrieveInputBuffer(request, minimumSize, ...) fails if the buffer is too small), tie the device's lifetime and cleanup to framework objects so unload ordering is automatic, and handle plug and play and power for hardware drivers. For any real driver, start with KMDF; you now know what it is doing underneath.

Common mistakes

Testing without Driver Verifier

Bugs that only appear under memory pressure or particular timing will otherwise reach users.

Skipping the hostile tests

Correct requests passing proves little; the incorrect ones are where drivers fail.

Running any of this outside the lab VM

Everything in this chapter is for the VM.

Nested, Inner and Anonymous Classes

lesson 7.011 of 5 in milestone 75 min2 practice tasks

Classes inside classes - static nested classes for helpers that belong to one class, inner classes that reach their outer object, local classes inside methods, anonymous classes, how lambdas replaced most of them, and the builder pattern.

A class can live inside another

When a class only makes sense together with another (a list's node, a map's entry, a builder), it can be declared inside it. Java has four kinds:

Kind Declared Has an outer object?
static nested class in a class, with static no
inner class in a class, without static yes
local class inside a method yes (if the method is not static)
anonymous class in an expression, with no name yes (if created in an instance method)

Static nested classes

A static nested class is an ordinary class that happens to be named through its outer class. It is the right choice most of the time:

Java
import java.util.ArrayList;
import java.util.List;

public class Main {
    static class Route {
        private final List<Stop> stops = new ArrayList<>();

        static class Stop {                       // Route.Stop
            final String name;
            final int minutes;
            Stop(String name, int minutes) { this.name = name; this.minutes = minutes; }
        }

        Route add(String name, int minutes) {
            stops.add(new Stop(name, minutes));
            return this;
        }

        int totalMinutes() {
            return stops.stream().mapToInt(s -> s.minutes).sum();
        }
    }

    public static void main(String[] args) {
        Route r = new Route().add("Depot", 0).add("Market", 12).add("Station", 7);
        Route.Stop first = r.stops.get(0);       // nested classes can see each other's private parts
        System.out.println(first.name + ", total " + r.totalMinutes() + " min");
    }
}

Inner classes

Without static, a nested class is inner: every object of it is tied to an object of the outer class and can use that object's fields directly. Iterators are the classic example: the iterator needs the collection it walks.

Java
import java.util.Iterator;
import java.util.NoSuchElementException;

public class Main {
    static class Countdown implements Iterable<Integer> {
        private final int start;
        Countdown(int start) { this.start = start; }

        public Iterator<Integer> iterator() {
            return new Steps();
        }

        private class Steps implements Iterator<Integer> {    // inner: sees start
            private int next = start;
            public boolean hasNext() { return next > 0; }
            public Integer next() {
                if (!hasNext()) throw new NoSuchElementException();
                return next--;
            }
        }
    }

    public static void main(String[] args) {
        for (int n : new Countdown(3)) System.out.print(n + " ");
        System.out.println("lift-off");
    }
}

Because an inner object keeps a hidden reference to its outer object, it keeps that object alive. Prefer static nested classes unless you really need the outer object.

Anonymous classes

An anonymous class declares and creates a one-off class in one expression. Before Java 8 it was the only way to pass behaviour around:

Java
import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;

public class Main {
    public static void main(String[] args) {
        List<String> words = new ArrayList<>(List.of("kiwi", "fig", "banana"));
        words.sort(new Comparator<String>() {            // an anonymous class
            @Override
            public int compare(String a, String b) {
                return Integer.compare(a.length(), b.length());
            }
        });
        System.out.println(words);

        words.sort(Comparator.comparing(String::length).reversed());   // the same idea today
        System.out.println(words);
    }
}

For an interface with one abstract method (a functional interface, lesson 6.01) a lambda is shorter and clearer. Anonymous classes are still useful when you need several methods, or state of their own.

Local classes and capture

A class (or lambda) declared inside a method can use the method's local variables, but only if they are effectively final (never changed after being set):

Java
public class Main {
    interface Greeter { String greet(String name); }

    static Greeter politeGreeter(String title) {
        class Polite implements Greeter {                // a local class
            public String greet(String name) { return "Good day, " + title + " " + name; }
        }
        return new Polite();
    }

    public static void main(String[] args) {
        System.out.println(politeGreeter("Dr.").greet("Hopper"));
    }
}

The builder pattern

A static nested Builder gives objects with many optional settings a readable way to be made, and keeps the finished object immutable:

Java
public class Main {
    static final class Email {
        private final String to, subject, body;
        private final boolean urgent;

        private Email(Builder b) { to = b.to; subject = b.subject; body = b.body; urgent = b.urgent; }

        static Builder to(String to) { return new Builder(to); }

        static final class Builder {
            private final String to;
            private String subject = "(no subject)", body = "";
            private boolean urgent;
            private Builder(String to) { this.to = to; }
            Builder subject(String s) { subject = s; return this; }
            Builder body(String b) { body = b; return this; }
            Builder urgent() { urgent = true; return this; }
            Email build() { return new Email(this); }
        }

        @Override public String toString() { return (urgent ? "[!] " : "") + to + ": " + subject; }
    }

    public static void main(String[] args) {
        Email e = Email.to("ada@example.com").subject("Notes").urgent().build();
        System.out.println(e);
    }
}

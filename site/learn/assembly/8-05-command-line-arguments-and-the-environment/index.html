Command-Line Arguments and the Environment

lesson 8.055 of 5 in milestone 85 min2 practice tasks

What a Linux program finds on its stack at _start - argc, the argv pointers and the zero that ends them, the environment strings after them, walking them in assembly, comparing a string prefix to find a variable such as HOME, turning arguments into numbers, and setting the exit code that scripts check.

The stack at _start

When Linux starts a program, the kernel puts information on the stack before the first instruction runs. At _start, rsp points at:

address holds
[rsp] argc, the number of arguments (including the program's name)
[rsp + 8] argv[0], a pointer to the program's name
[rsp + 16] argv[1], the first real argument
... ...
[rsp + 8 × (argc + 1)] 0, the end of argv
after that pointers to the environment strings "NAME=value", then 0

Each pointer leads to a zero-terminated string. _start is not a function (nothing called it), so there is no return address; read these before pushing anything.

Printing the arguments

Assembly
        global  _start

        section .data
count_msg: db   "arguments: ", 0

        section .text
_start:
        mov     r12, [rsp]              ; argc
        lea     r13, [rsp + 8]          ; &argv[0]

        lea     rdi, [rel count_msg]
        call    print_str
        mov     rdi, r12
        call    print_int
        call    print_newline

        xor     ebx, ebx
.next:  cmp     rbx, r12
        jge     .done
        mov     rdi, rbx
        call    print_int
        mov     edi, ':'
        call    print_char
        mov     edi, ' '
        call    print_char
        mov     rdi, [r13 + rbx*8]      ; argv[i]
        call    print_str
        call    print_newline
        inc     rbx
        jmp     .next
.done:
        mov     eax, 60
        xor     edi, edi
        syscall

On this site, the Run options let you give arguments; without any, argc is 1 and only the program's name is printed. In a terminal, ./program one "two words" gives argc = 3, and the quotes keep two words together as one argument.

The environment

The environment strings start right after the zero that ends argv. To find one variable, compare the start of each string with "NAME=":

Assembly
        global  _start

        section .data
want:   db      "PATH=", 0
label:  db      "PATH is: ", 0
none:   db      "not set", 0

        section .text
starts_with:                            ; rdi = text, rsi = prefix; rax = 1 if text starts with prefix
.loop:  movzx   eax, byte [rsi]
        test    al, al
        jz      .yes                    ; reached the end of the prefix
        cmp     al, [rdi]
        jne     .no
        inc     rdi
        inc     rsi
        jmp     .loop
.yes:   mov     eax, 1
        ret
.no:    xor     eax, eax
        ret

_start:
        mov     rax, [rsp]              ; argc
        lea     rbx, [rsp + 8 + rax*8 + 8]   ; skip argc, the argv pointers and their zero
.scan:  mov     r12, [rbx]
        test    r12, r12
        jz      .missing
        mov     rdi, r12
        lea     rsi, [rel want]
        call    starts_with
        test    eax, eax
        jnz     .found
        add     rbx, 8
        jmp     .scan
.found:
        lea     rdi, [rel label]
        call    print_str
        lea     rdi, [r12 + 5]          ; the value, just after "PATH="
        call    print_str
        call    print_newline
        jmp     .exit
.missing:
        lea     rdi, [rel none]
        call    print_str
        call    print_newline
.exit:  mov     eax, 60
        xor     edi, edi
        syscall

The runner on this site starts programs with an almost empty environment, so the value may be blank here; in a terminal you see your real PATH.

The comparison stops at the end of the prefix, so PATH= does not match a variable called PATHEXT= by accident, because = must match too.

Numbers from arguments

Arguments are always text. Parse them with the same digit loop as lesson 6.02:

Assembly
        global  _start

        section .data
sample: db      "-250", 0

        section .text
parse_int:                              ; rdi = string; rax = value, rdx = 1 if valid
        xor     eax, eax
        xor     ecx, ecx                ; 1 if negative
        cmp     byte [rdi], '-'
        jne     .digits
        mov     ecx, 1
        inc     rdi
.digits:
        movzx   edx, byte [rdi]
        test    dl, dl
        jz      .end
        sub     edx, '0'
        cmp     edx, 9
        ja      .bad                    ; not a digit
        imul    rax, rax, 10
        add     rax, rdx
        inc     rdi
        jmp     .digits
.end:   test    ecx, ecx
        jz      .ok
        neg     rax
.ok:    mov     edx, 1
        ret
.bad:   xor     eax, eax
        xor     edx, edx
        ret

_start:
        lea     rdi, [rel sample]
        call    parse_int
        mov     rdi, rax
        call    print_int               ; -250
        call    print_newline
        mov     eax, 60
        xor     edi, edi
        syscall

With real arguments, pass [r13 + 8] (argv[1]) instead of sample, after checking that argc is at least 2.

The exit code

The number in rdi for the exit system call is what the shell sees in $?. Scripts depend on it: 0 for success, anything else for failure. A tool that was called wrongly usually prints a short usage message to standard error (file descriptor 2) and exits with 2. Run this one here without arguments and the result shows the message and exit code 2:

Assembly
        global  _start

        section .data
usage:  db      "usage: tool NUMBER", 10
usage_len equ   $ - usage

        section .text
_start:
        cmp     qword [rsp], 2          ; argc < 2: no NUMBER given
        jge     .ok
        mov     eax, 1                  ; write(2, usage, usage_len)
        mov     edi, 2
        lea     rsi, [rel usage]
        mov     edx, usage_len
        syscall
        mov     eax, 60
        mov     edi, 2
        syscall
.ok:    mov     eax, 60
        xor     edi, edi
        syscall

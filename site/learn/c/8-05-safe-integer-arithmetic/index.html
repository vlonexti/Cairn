Safe Integer Arithmetic

lesson 8.055 of 6 in milestone 85 min2 practice tasks

The integer rules that cause real bugs - integer promotion of small types, the usual arithmetic conversions, signed and unsigned comparisons, size_t and loops that never end, overflow as undefined behaviour versus unsigned wrap-around, checking before you add or multiply, the compiler's overflow builtins and C23's ckd_add, and safe size calculations for malloc.

Small types are promoted

Before arithmetic, char and short values are promoted to int. That is usually harmless, but it explains some surprises:

C
#include <stdio.h>

int main(void)
{
    unsigned char a = 200, b = 100;
    int sum = a + b;                 /* 300: done in int, no wrap */
    unsigned char wrapped = a + b;   /* 44: wraps only when stored back */
    printf("%d %d\n", sum, wrapped);

    unsigned char x = 0x0F;
    printf("%d\n", (unsigned char)~x == 0xF0);   /* ~x is an int (0xFFFFFFF0), so cast back before comparing */
    return 0;
}

Mixing signed and unsigned

When an expression mixes a signed and an unsigned type of the same size, the signed one is converted to unsigned. Negative numbers become huge:

C
#include <stdio.h>
#include <string.h>

int main(void)
{
    int balance = -5;
    unsigned int limit = 10;
    if (balance < limit) printf("as expected\n");
    else printf("-5 < 10 is false here: -5 became %u\n", (unsigned)balance);

    const char *s = "hi";
    int wanted = -1;
    if (strlen(s) > wanted) printf("never printed\n");
    else printf("strlen(s) > -1 is false: -1 became a huge size_t\n");
    return 0;
}

Compile with -Wall -Wextra (this site does): GCC warns with -Wsign-compare. Keep counts and sizes in one unsigned type (size_t), and convert explicitly when you must compare with a signed value.

Loops that never end

C
#include <stdio.h>

int main(void)
{
    int data[] = {1, 2, 3};
    size_t n = 3;
    /* for (size_t i = n - 1; i >= 0; i--)  never ends: a size_t is always >= 0 */
    for (size_t i = n; i-- > 0;) printf("%d ", data[i]);     /* the safe countdown */
    printf("\n");
    return 0;
}

Overflow: signed versus unsigned

  • Unsigned arithmetic wraps around by definition: UINT_MAX + 1u == 0. That is well defined, but usually still a bug.
  • Signed overflow is undefined behaviour (lesson 6.02): the compiler may assume it never happens, and optimised code can do anything.

So the test must happen before the operation, never after:

C
#include <limits.h>
#include <stdbool.h>
#include <stdint.h>
#include <stdio.h>

static bool add_ok(int a, int b, int *out)
{
    if ((b > 0 && a > INT_MAX - b) || (b < 0 && a < INT_MIN - b)) return false;
    *out = a + b;
    return true;
}

static bool mul_size_ok(size_t a, size_t b, size_t *out)
{
    if (a != 0 && b > SIZE_MAX / a) return false;
    *out = a * b;
    return true;
}

int main(void)
{
    int r;
    printf("%d ", add_ok(INT_MAX - 1, 1, &r) ? r : -1);
    printf("%s\n", add_ok(INT_MAX, 1, &r) ? "ok" : "would overflow");
    size_t bytes;
    printf("%s\n", mul_size_ok(SIZE_MAX / 2, 3, &bytes) ? "ok" : "too big to allocate");
    return 0;
}

The wrong way, if (a + b < a), already overflowed; for signed types the compiler may even delete the check.

Builtins and C23

GCC and Clang have builtins that do the operation and report overflow, for any integer types:

C
#include <stdio.h>

int main(void)
{
    int r;
    if (__builtin_add_overflow(2000000000, 2000000000, &r)) printf("overflow\n");
    long long big;
    if (!__builtin_mul_overflow(3000000000LL, 3, &big)) printf("%lld\n", big);
    return 0;
}

C23 standardises the same idea in <stdckdint.h>: bool ckd_add(type *result, a, b) (and ckd_sub, ckd_mul), returning true on overflow.

Sizes for malloc

malloc(count * sizeof(T)) with a huge count from a file can overflow, allocate a tiny block, and let later writes run past it. Check the multiplication, or use calloc(count, sizeof(T)), which must check it for you:

C
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>

struct record { char name[32]; int score; };

static struct record *alloc_records(size_t count)
{
    if (count > SIZE_MAX / sizeof(struct record)) return NULL;
    return malloc(count * sizeof(struct record));
}

int main(void)
{
    struct record *a = alloc_records(100);
    struct record *b = alloc_records(SIZE_MAX / 4);
    printf("%s %s\n", a ? "allocated" : "refused", b ? "allocated" : "refused");
    free(a);
    free(b);
    return 0;
}

Converting between types

Converting a value to a type that cannot hold it:

  • to an unsigned type: wraps (well defined);
  • to a signed type: the result is implementation-defined (in practice it wraps), so check the range first;
  • from a floating-point value out of range to an integer: undefined behaviour.
C
#include <limits.h>
#include <stdio.h>

static int to_int_checked(long long v, int *out)
{
    if (v < INT_MIN || v > INT_MAX) return 0;
    *out = (int)v;
    return 1;
}

int main(void)
{
    int x;
    printf("%d %d\n", to_int_checked(123, &x), to_int_checked(5000000000LL, &x));
    return 0;
}

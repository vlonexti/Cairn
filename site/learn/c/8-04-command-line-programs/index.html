Command-Line Programs

lesson 8.044 of 6 in milestone 85 min2 practice tasks

Writing C tools that behave well in a terminal - reading options from argv by hand, flags with values, the conventions of -h, -- and combined short flags, getopt where it exists, standard input, output and error, exit codes with EXIT_SUCCESS and EXIT_FAILURE, reading lines of any length, and working in pipelines.

What a good command-line tool does

Unix-style tools are small programs that fit together. They follow conventions so people (and scripts) can use them without reading the source:

  • options start with - (short, -n 5) or -- (long, --lines=5); -h or --help prints how to use it;
  • input comes from files named on the command line, or standard input when there are none;
  • results go to standard output, messages and errors to standard error;
  • the exit code is 0 for success and something else for failure.

Reading options by hand

C
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

struct options {
    int lines;
    int number;
    const char *file;
};

static int parse_args(int argc, char **argv, struct options *o)
{
    o->lines = 10;
    o->number = 0;
    o->file = NULL;
    for (int i = 1; i < argc; i++) {
        const char *a = argv[i];
        if (strcmp(a, "-n") == 0 || strcmp(a, "--lines") == 0) {
            if (i + 1 >= argc) { fprintf(stderr, "%s needs a number\n", a); return -1; }
            char *end;
            long v = strtol(argv[++i], &end, 10);
            if (*end != '\0' || v < 0) { fprintf(stderr, "bad number: %s\n", argv[i]); return -1; }
            o->lines = (int)v;
        } else if (strncmp(a, "--lines=", 8) == 0) {
            o->lines = atoi(a + 8);
        } else if (strcmp(a, "-N") == 0) {
            o->number = 1;
        } else if (strcmp(a, "--") == 0) {
            if (i + 1 < argc) o->file = argv[i + 1];
            break;
        } else if (a[0] == '-' && a[1] != '\0') {
            fprintf(stderr, "unknown option %s\n", a);
            return -1;
        } else {
            o->file = a;
        }
    }
    return 0;
}

int main(void)
{
    char *demo[] = {"head", "-n", "3", "-N", "notes.txt"};
    struct options o;
    if (parse_args(5, demo, &o) != 0) return EXIT_FAILURE;
    printf("lines %d, number %d, file %s\n", o.lines, o.number, o.file ? o.file : "(stdin)");
    return EXIT_SUCCESS;
}

-- means "everything after this is a file name, even if it starts with -". A lone - is often a name for standard input.

getopt

POSIX systems (Linux, macOS) have getopt in <unistd.h>, which handles short options, combined flags like -Nv, and values like -n5 or -n 5:

C
#include <unistd.h>

int c;
while ((c = getopt(argc, argv, "n:Nh")) != -1) {   /* n takes a value (the colon) */
    switch (c) {
    case 'n': lines = atoi(optarg); break;
    case 'N': number = 1; break;
    case 'h': usage(stdout); return 0;
    default:  usage(stderr); return 2;               /* getopt already printed the problem */
    }
}
/* argv[optind] .. argv[argc - 1] are the remaining file names */

It is not part of standard C, so Windows programs parse by hand or bring their own.

Three streams

C
#include <stdio.h>
#include <stdlib.h>

int main(void)
{
    printf("results go to standard output\n");
    fprintf(stderr, "warnings go to standard error\n");
    return EXIT_SUCCESS;
}

With tool > out.txt, only standard output goes into the file, and errors still appear on screen. Never print error messages to stdout: they would end up mixed into data another program reads.

Reading lines of any length

fgets with a fixed buffer splits a long line into pieces. To read whole lines of any length, grow a buffer (lesson 8.01's idea):

C
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* Reads one line without its '\n'. Returns a new string the caller frees, or NULL at the end of the input. */
static char *read_line(FILE *f)
{
    size_t cap = 32, len = 0;
    char *buf = malloc(cap);
    if (!buf) return NULL;
    int c;
    while ((c = fgetc(f)) != EOF && c != '\n') {
        if (len + 1 >= cap) {
            char *p = realloc(buf, cap *= 2);
            if (!p) { free(buf); return NULL; }
            buf = p;
        }
        buf[len++] = (char)c;
    }
    if (c == EOF && len == 0) { free(buf); return NULL; }
    if (len > 0 && buf[len - 1] == '\r') len--;       /* Windows line endings */
    buf[len] = '\0';
    return buf;
}

int main(void)
{
    FILE *f = tmpfile();
    if (!f) return EXIT_FAILURE;
    fputs("short\na much longer line that would not fit in a small fixed buffer at all\r\nlast", f);
    rewind(f);
    char *line;
    int n = 0;
    while ((line = read_line(f)) != NULL) {
        printf("%d: %zu characters\n", ++n, strlen(line));
        free(line);
    }
    fclose(f);
    return EXIT_SUCCESS;
}

POSIX has getline(&buf, &cap, f), which does the same.

Exit codes

return from main (or exit(code)) sets the exit code. EXIT_SUCCESS (0) and EXIT_FAILURE (1) are the portable values. Many tools use 2 for "you called me wrongly" and 1 for "something failed while working". Scripts depend on them:

Text
$ mytool data.txt && echo ok || echo failed

Being a good pipeline citizen

  • Read standard input when no file is given, so cat data.txt | mytool and mytool < data.txt work.
  • Write nothing but results to standard output.
  • Process input as a stream (line by line) instead of loading everything, so huge inputs work.
  • Check the result of writing too: if (ferror(stdout)) after printing reports a full disk or a closed pipe.

Dynamic Strings

lesson 8.011 of 6 in milestone 85 min2 practice tasks

Building text of any length safely in C - why fixed char arrays run out, a growable string buffer with length and capacity, appending with realloc, a printf-style append with vsnprintf that measures before it writes, handing ownership of the result to the caller, and the common string-building bugs this design prevents.

Fixed buffers run out

char line[100] works until someone's name, path or message is longer than you planned. strcat into a fixed buffer then writes past its end, which is one of the classic security bugs. Programs that build text of unknown length need a buffer that grows.

A string buffer

Keep three things together: the bytes, how many are used, and how many are allocated:

C
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    char *data;     /* always zero-terminated when not NULL */
    size_t len;     /* characters used, not counting the zero */
    size_t cap;     /* bytes allocated */
} StrBuf;

static int sb_reserve(StrBuf *sb, size_t extra)
{
    size_t need = sb->len + extra + 1;
    if (need <= sb->cap) return 0;
    size_t cap = sb->cap ? sb->cap : 16;
    while (cap < need) cap *= 2;
    char *p = realloc(sb->data, cap);
    if (!p) return -1;               /* the old block is still valid */
    sb->data = p;
    sb->cap = cap;
    return 0;
}

static int sb_append(StrBuf *sb, const char *text)
{
    size_t n = strlen(text);
    if (sb_reserve(sb, n) != 0) return -1;
    memcpy(sb->data + sb->len, text, n + 1);   /* copies the zero too */
    sb->len += n;
    return 0;
}

int main(void)
{
    StrBuf sb = {0};
    for (int i = 0; i < 5; i++) {
        sb_append(&sb, "word ");
    }
    printf("\"%s\" len %zu cap %zu\n", sb.data, sb.len, sb.cap);
    free(sb.data);
    return 0;
}

The capacity doubles, so appending n characters one at a time costs only O(n) copying overall (the same idea as std::vector and lesson 4.02's growing array).

Starting from {0} (all zero) means "empty, nothing allocated", and free(NULL) is allowed, so an unused buffer needs no special case.

Appending formatted text

snprintf can tell you how long its output would be without writing it, by passing a size of 0. That makes a safe printf-style append: measure, grow, then write.

C
#include <stdarg.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct { char *data; size_t len, cap; } StrBuf;

static int sb_reserve(StrBuf *sb, size_t extra)
{
    size_t need = sb->len + extra + 1;
    if (need <= sb->cap) return 0;
    size_t cap = sb->cap ? sb->cap : 16;
    while (cap < need) cap *= 2;
    char *p = realloc(sb->data, cap);
    if (!p) return -1;
    sb->data = p;
    sb->cap = cap;
    return 0;
}

static int sb_appendf(StrBuf *sb, const char *fmt, ...)
{
    va_list ap, copy;
    va_start(ap, fmt);
    va_copy(copy, ap);
    int n = vsnprintf(NULL, 0, fmt, copy);      /* how long would it be? */
    va_end(copy);
    if (n < 0 || sb_reserve(sb, (size_t)n) != 0) {
        va_end(ap);
        return -1;
    }
    vsnprintf(sb->data + sb->len, sb->cap - sb->len, fmt, ap);
    va_end(ap);
    sb->len += (size_t)n;
    return 0;
}

int main(void)
{
    StrBuf sb = {0};
    const char *names[] = {"Ada", "Grace", "Alan"};
    for (int i = 0; i < 3; i++) {
        sb_appendf(&sb, "%s%s scored %d", i ? ", " : "", names[i], 90 - i * 5);
    }
    puts(sb.data);
    free(sb.data);
    return 0;
}

A va_list can only be walked once, so va_copy makes a second one for the measuring call (lesson 7.01).

Who frees it?

A function that builds a string and returns it gives the caller ownership: the caller must free it. Say so in the comment, every time:

C
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* Returns a new string "first last" that the caller must free, or NULL if out of memory. */
static char *full_name(const char *first, const char *last)
{
    size_t n = strlen(first) + 1 + strlen(last) + 1;
    char *out = malloc(n);
    if (!out) return NULL;
    snprintf(out, n, "%s %s", first, last);
    return out;
}

int main(void)
{
    char *name = full_name("Ada", "Lovelace");
    if (!name) return 1;
    printf("%s (%zu characters)\n", name, strlen(name));
    free(name);
    return 0;
}

The other common style is that the caller supplies the buffer and its size (snprintf's own design), and the function reports how much it needed. Both are fine; mixing them up is how leaks and double frees happen.

Taking the result out

A buffer can hand its bytes over and reset itself, so the result outlives the buffer:

C
char *sb_take(StrBuf *sb)            /* the caller now owns the returned string */
{
    char *s = sb->data ? sb->data : calloc(1, 1);   /* an empty string, never NULL */
    sb->data = NULL;
    sb->len = sb->cap = 0;
    return s;
}

Bugs this prevents

  • Writing past the end: every write checks the capacity first.
  • Off-by-one on the zero: sb_reserve always asks for one byte more.
  • Losing the block when realloc fails: the result goes into a temporary before replacing data.
  • Quadratic time: strcat scans the whole string to find its end each time; keeping len avoids that.

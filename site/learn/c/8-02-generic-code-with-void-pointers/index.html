Generic Code with void Pointers

lesson 8.022 of 6 in milestone 85 min2 practice tasks

Writing one container or algorithm for every type in C - void pointers and element sizes, copying with memcpy, a generic swap, a growable vector that stores any element type, comparison callbacks as qsort and bsearch use them, the type safety you give up, and macros that give some of it back.

One function for every type

C has no templates or generics. Code that works for any type uses void *, a pointer to "some bytes", together with the size of one element. The standard library's qsort, bsearch, memcpy and memcmp all work this way.

A generic swap

C
#include <stdio.h>
#include <string.h>

static void swap_any(void *a, void *b, size_t size)
{
    unsigned char tmp[64];
    if (size > sizeof tmp) return;            /* this version handles elements up to 64 bytes */
    memcpy(tmp, a, size);
    memcpy(a, b, size);
    memcpy(b, tmp, size);
}

struct point { double x, y; };

int main(void)
{
    int i = 1, j = 2;
    swap_any(&i, &j, sizeof i);
    struct point p = {1, 2}, q = {3, 4};
    swap_any(&p, &q, sizeof p);
    printf("%d %d | %.0f,%.0f %.0f,%.0f\n", i, j, p.x, p.y, q.x, q.y);
    return 0;
}

The function never knows what it is swapping; it only moves bytes. memcpy is the safe way to copy an object's bytes regardless of type.

Pointer arithmetic on bytes

Arithmetic on a void * is not allowed in standard C (it has no size). Cast to unsigned char * (or char *) and step in bytes:

C
#include <stdio.h>

static void *element(void *base, size_t index, size_t size)
{
    return (unsigned char *)base + index * size;
}

int main(void)
{
    double values[] = {1.5, 2.5, 3.5};
    double *second = element(values, 1, sizeof values[0]);
    printf("%.1f\n", *second);
    return 0;
}

A vector for any type

C
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    unsigned char *data;
    size_t len, cap, elem;      /* elem: the size of one element in bytes */
} Vec;

static Vec vec_new(size_t elem) { Vec v = {NULL, 0, 0, elem}; return v; }

static int vec_push(Vec *v, const void *item)
{
    if (v->len == v->cap) {
        size_t cap = v->cap ? v->cap * 2 : 4;
        unsigned char *p = realloc(v->data, cap * v->elem);
        if (!p) return -1;
        v->data = p;
        v->cap = cap;
    }
    memcpy(v->data + v->len * v->elem, item, v->elem);
    v->len++;
    return 0;
}

static void *vec_at(const Vec *v, size_t i) { return i < v->len ? v->data + i * v->elem : NULL; }

struct player { char name[16]; int score; };

int main(void)
{
    Vec scores = vec_new(sizeof(int));
    for (int i = 1; i <= 10; i++) { int sq = i * i; vec_push(&scores, &sq); }
    printf("%zu ints, the 5th is %d\n", scores.len, *(int *)vec_at(&scores, 4));

    Vec players = vec_new(sizeof(struct player));
    struct player a = {"Ada", 91}, b = {"Bob", 48};
    vec_push(&players, &a);
    vec_push(&players, &b);
    struct player *p = vec_at(&players, 1);
    printf("%s %d, out of range: %p\n", p->name, p->score, vec_at(&players, 9));

    free(scores.data);
    free(players.data);
    return 0;
}

The vector stores copies of the elements. Storing pointers instead (sizeof(struct player *)) is also common, but then someone else must own and free the pointed-to objects.

Callbacks that compare

qsort and bsearch take a comparison function with the signature int (*)(const void *, const void *). It returns negative, zero or positive, like strcmp:

C
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

struct player { const char *name; int score; };

static int by_score_desc(const void *a, const void *b)
{
    const struct player *x = a, *y = b;
    return (y->score > x->score) - (y->score < x->score);
}

static int by_name(const void *a, const void *b)
{
    const struct player *x = a, *y = b;
    return strcmp(x->name, y->name);
}

int main(void)
{
    struct player ps[] = {{"Cy", 77}, {"Ada", 91}, {"Bob", 48}, {"Dee", 91}};
    size_t n = sizeof ps / sizeof ps[0];

    qsort(ps, n, sizeof ps[0], by_score_desc);
    for (size_t i = 0; i < n; i++) printf("%s %d  ", ps[i].name, ps[i].score);
    printf("\n");

    qsort(ps, n, sizeof ps[0], by_name);
    struct player key = {"Cy", 0};
    struct player *found = bsearch(&key, ps, n, sizeof ps[0], by_name);
    printf("found %s with %d\n", found ? found->name : "nobody", found ? found->score : 0);
    return 0;
}

(y > x) - (y < x) gives -1, 0 or 1 without the overflow risk of subtracting. bsearch needs the array sorted by the same comparison it searches with.

The cost: no type checking

With void *, the compiler cannot catch vec_push(&scores, &some_double): it copies 4 bytes of a double into an int slot, and you get nonsense. Defences:

  • keep each generic container used for one type only, with a clear name (Vec scores; /* of int */);
  • wrap it in small typed functions or macros:
C
#include <stdio.h>
#include <stdlib.h>

#define DEFINE_STACK(T, NAME)                                                    \
    typedef struct { T *items; size_t len, cap; } NAME;                          \
    static int NAME##_push(NAME *s, T value)                                     \
    {                                                                            \
        if (s->len == s->cap) {                                                  \
            size_t cap = s->cap ? s->cap * 2 : 4;                                \
            T *p = realloc(s->items, cap * sizeof *p);                           \
            if (!p) return -1;                                                   \
            s->items = p;                                                        \
            s->cap = cap;                                                        \
        }                                                                        \
        s->items[s->len++] = value;                                              \
        return 0;                                                                \
    }

DEFINE_STACK(double, DoubleStack)

int main(void)
{
    DoubleStack s = {0};
    DoubleStack_push(&s, 1.5);
    DoubleStack_push(&s, 2.5);
    printf("%zu items, top %.1f\n", s.len, s.items[s.len - 1]);
    free(s.items);
    return 0;
}

The macro writes a real, typed stack for each type you ask for, so the compiler checks every push. It is how many C libraries offer "generic" containers with type safety.
